       ```typescript
       private static escapeCsvField(value: string | number | null | undefined): string {
         if (value == null) return ''
         const str = String(value)
         // Escape if contains comma, quote, or newline
         if (str.includes(',') || str.includes('"') || str.includes('\n')) {
           return `"${str.replace(/"/g, '""')}"`
         }
         return str
       }
       ```

       ```typescript
       const handleExport = async (type: string, filename: string) => {
         try {
           let result
           if (type === 'assets') {
             const assetsWithRelations = await loadAssetsWithRelations() // Load asset data with type/location names
             result = await csvExportService.exportAssets(assetsWithRelations, filename)
           } else if (type === 'locations') {
             const locationsWithParents = await loadLocationsWithParents() // Load location data with parent names
             result = await csvExportService.exportLocations(locationsWithParents, filename)

       }
       ```
