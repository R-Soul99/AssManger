Create CSV export service with proper UTF-8 BOM and field escaping:

**Field escaping function:**
```typescript
function escapeCsvField(value: string | number | boolean | null | undefined): string {
  if (value == null) return '';
  const str = String(value);

  // If contains comma, quote, or newline, wrap in quotes and escape internal quotes
  if (str.includes(',') || str.includes('"') || str.includes('\n') || str.includes('\r')) {
    return `"${str.replace(/"/g, '""')}"`;
  }
  return str;
}
```

**Service class:**
```typescript
import { save } from '@tauri-apps/plugin-dialog';
import { writeFile } from '@tauri-apps/plugin-fs';
import { revealItemInDir } from '@tauri-apps/plugin-opener';

export class CsvExportService {
  private static readonly UTF8_BOM = '\uFEFF';

  /**
   * Export assets to CSV
   */
  async exportAssets(
    assets: AssetWithRelations[],
    filename: string = 'assets.csv'
  ): Promise<{ success: true; path: string } | { success: false; error: string }> {
    const headers = [
      'Asset Tag', 'Description', 'Category', 'Location',
      'Status', 'Serial Number', 'Phone/Extension', 'Owner',
      'Cost Centre', 'Cost', 'Purchase Date', 'Created', 'Updated'
    ];

    const rows = assets.map(({ asset, category, locationPath }) => [
      escapeCsvField(asset.tag),
      escapeCsvField(asset.description),
      escapeCsvField(category?.name),
      escapeCsvField(locationPath),
      escapeCsvField(asset.status),
      escapeCsvField(asset.serialNumber),
      escapeCsvField(asset.phoneExtension),
      escapeCsvField(asset.owner),
      escapeCsvField(asset.costCentre),
      escapeCsvField(asset.cost),
      escapeCsvField(asset.purchaseDate?.toISOString().split('T')[0]),  // YYYY-MM-DD
      escapeCsvField(asset.createdAt.toISOString()),
      escapeCsvField(asset.updatedAt.toISOString()),
    ]);

    return this.writeCSV(headers, rows, filename);
  }

  /**
   * Export locations to CSV
   */
  async exportLocations(
    locations: LocationData[],
    filename: string = 'locations.csv'
  ): Promise<{ success: true; path: string } | { success: false; error: string }> {
    const headers = ['ID', 'Name', 'Type', 'Parent ID', 'Description', 'Created', 'Updated'];

    const rows = locations.map(loc => [
      escapeCsvField(loc.id),
      escapeCsvField(loc.name),
      escapeCsvField(loc.type),
      escapeCsvField(loc.parentId),
      escapeCsvField(loc.description),
      escapeCsvField(loc.createdAt.toISOString()),
      escapeCsvField(loc.updatedAt.toISOString()),
    ]);

    return this.writeCSV(headers, rows, filename);
  }

  /**
   * Export categories to CSV
   */
  async exportCategories(
    categories: CategoryData[],
    filename: string = 'categories.csv'
  ): Promise<{ success: true; path: string } | { success: false; error: string }> {
    const headers = ['ID', 'Name', 'Description', 'Icon', 'Color', 'Parent ID', 'Created', 'Updated'];

    const rows = categories.map(cat => [
      escapeCsvField(cat.id),
      escapeCsvField(cat.name),
      escapeCsvField(cat.description),
      escapeCsvField(cat.icon),
      escapeCsvField(cat.color),
      escapeCsvField(cat.parentId),
      escapeCsvField(cat.createdAt.toISOString()),
      escapeCsvField(cat.updatedAt.toISOString()),
    ]);

    return this.writeCSV(headers, rows, filename);
  }

  /**
   * Write CSV content to file with save dialog
   */
  private async writeCSV(
    headers: string[],
    rows: string[][],
    defaultFilename: string
  ): Promise<{ success: true; path: string } | { success: false; error: string }> {
    try {
      // Build CSV content with UTF-8 BOM
      const csvContent = CsvExportService.UTF8_BOM +
        headers.join(',') + '\n' +
        rows.map(row => row.join(',')).join('\n');

      // Show save dialog
      const filePath = await save({
        defaultPath: defaultFilename,
        filters: [{ name: 'CSV Files', extensions: ['csv'] }],
      });

      if (!filePath) {
        return { success: false, error: 'Export cancelled' };
      }

      // Write file
      const encoder = new TextEncoder();
      await writeFile(filePath, encoder.encode(csvContent));

      // Reveal in Explorer
      await revealItemInDir(filePath);

      return { success: true, path: filePath };
    } catch (error) {
      return {
        success: false,
        error: `Export failed: ${error instanceof Error ? error.message : 'Unknown error'}`
      };
    }
  }
}
```

```tsx
const ExportDialog: React.FC<ExportDialogProps> = ({
  open, onClose, filteredAssetCount, totalAssetCount,
  onExportAssets, onExportLocations, onExportCategories
}) => {
  const [assetScope, setAssetScope] = useState<'filtered' | 'all'>('filtered');
  const [exporting, setExporting] = useState(false);
  const [message, setMessage] = useState<{ type: 'success' | 'error'; text: string } | null>(null);

  const handleExportAssets = async () => {
    setExporting(true);
    setMessage(null);
    try {
      await onExportAssets(assetScope === 'all');
      setMessage({ type: 'success', text: 'Assets exported successfully' });
    } catch (e) {
      setMessage({ type: 'error', text: e.message });
    }
    setExporting(false);
  };

  const handleExportLocations = async () => {
    setExporting(true);
    setMessage(null);
    try {
      await onExportLocations();
      setMessage({ type: 'success', text: 'Locations exported successfully' });
    } catch (e) {
      setMessage({ type: 'error', text: e.message });
    }
    setExporting(false);
  };

  const handleExportCategories = async () => {
    setExporting(true);
    setMessage(null);
    try {
      await onExportCategories();
      setMessage({ type: 'success', text: 'Categories exported successfully' });
    } catch (e) {
      setMessage({ type: 'error', text: e.message });
    }
    setExporting(false);
  };

  return (
    <Dialog open={open} onClose={onClose} maxWidth="sm" fullWidth>
      <DialogTitle>Export Data</DialogTitle>
      <DialogContent>
        {message && (
          <Alert severity={message.type} sx={{ mb: 2 }}>
            {message.text}
          </Alert>
        )}

        <Typography variant="subtitle1" gutterBottom>Assets</Typography>
        <RadioGroup value={assetScope} onChange={(e) => setAssetScope(e.target.value)}>
          <FormControlLabel
            value="filtered"
            control={<Radio />}
            label={`Export Filtered (${filteredAssetCount} assets)`}
          />
          <FormControlLabel
            value="all"
            control={<Radio />}
            label={`Export All (${totalAssetCount} assets)`}
          />
        </RadioGroup>
        <Button
          variant="contained"
          onClick={handleExportAssets}
          disabled={exporting}
          sx={{ mt: 1, mb: 2 }}
        >
          Export Assets
        </Button>

        <Divider sx={{ my: 2 }} />

        <Typography variant="subtitle1" gutterBottom>Other Data</Typography>
        <Stack direction="row" spacing={2}>
          <Button
            variant="outlined"
            onClick={handleExportLocations}
            disabled={exporting}
          >
            Export Locations
          </Button>
          <Button
            variant="outlined"
            onClick={handleExportCategories}
            disabled={exporting}
          >
            Export Categories
          </Button>
        </Stack>
      </DialogContent>
      <DialogActions>
        <Button onClick={onClose}>Close</Button>
      </DialogActions>
    </Dialog>
  );
};
```
