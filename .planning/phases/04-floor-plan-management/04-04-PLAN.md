```typescript
import {
  Dialog,
  DialogTitle,
  DialogContent,
  DialogActions,
  Button,
  Typography,
  Alert,
  CircularProgress,
  Box,
} from '@mui/material';
import { WarningAmber as WarningIcon } from '@mui/icons-material';
import { useState } from 'react';
import { FloorPlan } from '@/domain/entities';
import { FloorPlanService } from '@/application/services';
import { RepositoryFactory } from '@/infrastructure/repositories/RepositoryFactory';
import { localFileStorage } from '@/infrastructure/storage/LocalFileStorage';

interface FloorPlanDeleteDialogProps {
  open: boolean;
  floorPlan: FloorPlan | null;
  markerCount: number;
  onClose: () => void;
  onDeleted: () => void;
}

export function FloorPlanDeleteDialog({
  open,
  floorPlan,
  markerCount,
  onClose,
  onDeleted,
}: FloorPlanDeleteDialogProps) {
  const [deleting, setDeleting] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const handleDelete = async () => {
    if (!floorPlan) return;

    setDeleting(true);
    setError(null);

    try {
      const floorPlanService = new FloorPlanService(
        RepositoryFactory.getInstance().getFloorPlanRepository()
      );

      // Delete database record (this cascades to markers)
      const result = await floorPlanService.deleteFloorPlan(floorPlan.id);

      if (!result.success) {
        throw new Error(result.error);
      }

      // Delete image file
      try {
        await localFileStorage.deleteImage(floorPlan.imageRelativePath);
      } catch {
        // Image deletion failure is non-critical - log but don't fail
        console.warn('Failed to delete floor plan image file');
      }

      onDeleted();
      onClose();
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Failed to delete floor plan');
    } finally {
      setDeleting(false);
    }
  };

  if (!floorPlan) return null;

  return (
    <Dialog open={open} onClose={onClose} maxWidth="sm" fullWidth>
      <DialogTitle>Delete Floor Plan</DialogTitle>
      <DialogContent>
        {error && (
          <Alert severity="error" sx={{ mb: 2 }}>
            {error}
          </Alert>
        )}

        <Typography variant="body1" sx={{ mb: 2 }}>
          Are you sure you want to delete <strong>{floorPlan.name}</strong>?
        </Typography>

        {markerCount > 0 && (
          <Alert severity="warning" icon={<WarningIcon />} sx={{ mb: 2 }}>
            <Typography variant="body2">
              This floor plan has {markerCount} asset marker{markerCount !== 1 ? 's' : ''} placed on it.
              Deleting will remove {markerCount === 1 ? 'this marker' : 'these markers'}.
            </Typography>
          </Alert>
        )}

        <Typography variant="body2" color="text.secondary">
          This action cannot be undone. The floor plan image and all associated data will be permanently removed.
        </Typography>
      </DialogContent>
      <DialogActions>
        <Button onClick={onClose} disabled={deleting}>
          Cancel
        </Button>
        <Button
          onClick={handleDelete}
          color="error"
          variant="contained"
          disabled={deleting}
        >
          {deleting ? (
            <Box sx={{ display: 'flex', alignItems: 'center', gap: 1 }}>
              <CircularProgress size={16} color="inherit" />
              Deleting...
            </Box>
          ) : (
            'Delete'
          )}
        </Button>
      </DialogActions>
    </Dialog>
  );
}
```

Create FloorPlanBulkActions.tsx with stepped per-plan delete warnings:

```typescript
import { useState, useEffect } from 'react';
import {
  Box,
  Button,
  Typography,
  Dialog,
  DialogTitle,
  DialogContent,
  DialogActions,
  Alert,
  CircularProgress,
  LinearProgress,
} from '@mui/material';
import { Delete as DeleteIcon, WarningAmber as WarningIcon } from '@mui/icons-material';
import { FloorPlan } from '@/domain/entities';
import { FloorPlanService } from '@/application/services';
import { RepositoryFactory } from '@/infrastructure/repositories/RepositoryFactory';
import { localFileStorage } from '@/infrastructure/storage/LocalFileStorage';

interface FloorPlanBulkActionsProps {
  selectedIds: string[];
  floorPlans: FloorPlan[];
  markerCounts: Record<string, number>;
  onSelectionClear: () => void;
  onDeleted: () => void;
}

type DeletePhase =
  | { step: 'idle' }
  | { step: 'confirm-bulk'; count: number }
  | { step: 'warn-markers'; planIndex: number; plans: { plan: FloorPlan; count: number }[] }
  | { step: 'deleting'; progress: number; total: number }
  | { step: 'complete'; deleted: number; skipped: number };

export function FloorPlanBulkActions({
  selectedIds,
  floorPlans,
  markerCounts,
  onSelectionClear,
  onDeleted,
}: FloorPlanBulkActionsProps) {
  const [phase, setPhase] = useState<DeletePhase>({ step: 'idle' });
  const [confirmedForDelete, setConfirmedForDelete] = useState<string[]>([]);
  const [error, setError] = useState<string | null>(null);

  const selectedPlans = floorPlans.filter((p) => selectedIds.includes(p.id));

  // Reset state when selection changes
  useEffect(() => {
    setPhase({ step: 'idle' });
    setConfirmedForDelete([]);
    setError(null);
  }, [selectedIds.length]);

  const handleBulkDelete = () => {
    if (selectedIds.length === 0) return;
    setPhase({ step: 'confirm-bulk', count: selectedIds.length });
  };

  const handleConfirmBulk = () => {
    // Separate plans with and without markers
    const plansWithMarkers = selectedPlans
      .filter((p) => (markerCounts[p.id] || 0) > 0)
      .map((plan) => ({ plan, count: markerCounts[plan.id] || 0 }));

    const plansWithoutMarkers = selectedPlans.filter(
      (p) => (markerCounts[p.id] || 0) === 0
    );

    // Plans without markers are auto-confirmed
    const autoConfirmed = plansWithoutMarkers.map((p) => p.id);

    if (plansWithMarkers.length === 0) {
      // No marker warnings needed, proceed to delete
      setConfirmedForDelete(autoConfirmed);
      performDeletion(autoConfirmed);
    } else {
      // Step through plans with markers
      setConfirmedForDelete(autoConfirmed);
      setPhase({ step: 'warn-markers', planIndex: 0, plans: plansWithMarkers });
    }
  };

  const handleMarkerWarningResponse = (action: 'delete' | 'skip' | 'cancel') => {
    if (phase.step !== 'warn-markers') return;

    const { planIndex, plans } = phase;
    const currentPlan = plans[planIndex];

    if (action === 'cancel') {
      // Cancel entire operation
      setPhase({ step: 'idle' });
      setConfirmedForDelete([]);
      return;
    }

    if (action === 'delete') {
      // Add to confirmed list
      setConfirmedForDelete((prev) => [...prev, currentPlan.plan.id]);
    }
    // 'skip' does nothing to confirmed list

    // Move to next plan or proceed to deletion
    if (planIndex < plans.length - 1) {
      setPhase({ step: 'warn-markers', planIndex: planIndex + 1, plans });
    } else {
      // All warnings processed, proceed to deletion
      const finalConfirmed =
        action === 'delete'
          ? [...confirmedForDelete, currentPlan.plan.id]
          : confirmedForDelete;
      performDeletion(finalConfirmed);
    }
  };

  const performDeletion = async (idsToDelete: string[]) => {
    if (idsToDelete.length === 0) {
      setPhase({ step: 'complete', deleted: 0, skipped: selectedIds.length });
      return;
    }

    setPhase({ step: 'deleting', progress: 0, total: idsToDelete.length });

    const floorPlanService = new FloorPlanService(
      RepositoryFactory.getInstance().getFloorPlanRepository()
    );

    let deleted = 0;
    let errorOccurred = false;

    // Sequential delete loop (SQLite write safety per 03-03 decision)
    for (let i = 0; i < idsToDelete.length; i++) {
      const id = idsToDelete[i];
      const plan = floorPlans.find((p) => p.id === id);

      try {
        const result = await floorPlanService.deleteFloorPlan(id);
        if (result.success) {
          // Try to delete image file
          if (plan) {
            try {
              await localFileStorage.deleteImage(plan.imageRelativePath);
            } catch {
              // Non-critical
            }
          }
          deleted++;
        }
      } catch (err) {
        errorOccurred = true;
        setError(err instanceof Error ? err.message : 'Delete failed');
      }

      setPhase({
        step: 'deleting',
        progress: i + 1,
        total: idsToDelete.length,
      });
    }

    const skipped = selectedIds.length - deleted;
    setPhase({ step: 'complete', deleted, skipped });

    if (!errorOccurred) {
      // Auto-close and refresh after short delay
      setTimeout(() => {
        onDeleted();
        onSelectionClear();
        setPhase({ step: 'idle' });
      }, 1500);
    }
  };

  const handleClose = () => {
    setPhase({ step: 'idle' });
    setConfirmedForDelete([]);
    setError(null);
    if (phase.step === 'complete') {
      onDeleted();
      onSelectionClear();
    }
  };

  if (selectedIds.length === 0) return null;

  return (
    <>
      <Box
        sx={{
          display: 'flex',
          alignItems: 'center',
          gap: 2,
          p: 2,
          backgroundColor: 'primary.light',
          color: 'primary.contrastText',
          borderRadius: 1,
          mb: 2,
        }}
      >
        <Typography>
          {selectedIds.length} floor plan{selectedIds.length !== 1 ? 's' : ''} selected
        </Typography>
        <Button
          variant="contained"
          color="error"
          size="small"
          startIcon={<DeleteIcon />}
          onClick={handleBulkDelete}
        >
          Delete Selected
        </Button>
        <Button
          variant="outlined"
          size="small"
          onClick={onSelectionClear}
          sx={{ color: 'inherit', borderColor: 'inherit' }}
        >
          Clear Selection
        </Button>
      </Box>

      {/* Bulk confirm dialog */}
      <Dialog
        open={phase.step === 'confirm-bulk'}
        onClose={handleClose}
        maxWidth="sm"
        fullWidth
      >
        <DialogTitle>Delete Floor Plans</DialogTitle>
        <DialogContent>
          <Typography>
            Are you sure you want to delete {phase.step === 'confirm-bulk' ? phase.count : 0} floor plan
            {phase.step === 'confirm-bulk' && phase.count !== 1 ? 's' : ''}?
          </Typography>
        </DialogContent>
        <DialogActions>
          <Button onClick={handleClose}>Cancel</Button>
          <Button onClick={handleConfirmBulk} color="error" variant="contained">
            Continue
          </Button>
        </DialogActions>
      </Dialog>

      {/* Per-plan marker warning dialog */}
      <Dialog
        open={phase.step === 'warn-markers'}
        onClose={() => handleMarkerWarningResponse('cancel')}
        maxWidth="sm"
        fullWidth
      >
        <DialogTitle>
          Marker Warning ({phase.step === 'warn-markers' ? phase.planIndex + 1 : 0} of{' '}
          {phase.step === 'warn-markers' ? phase.plans.length : 0})
        </DialogTitle>
        <DialogContent>
          {phase.step === 'warn-markers' && (
            <>
              <Typography variant="body1" sx={{ mb: 2 }}>
                <strong>{phase.plans[phase.planIndex].plan.name}</strong>
              </Typography>
              <Alert severity="warning" icon={<WarningIcon />}>
                This floor plan has {phase.plans[phase.planIndex].count} asset marker
                {phase.plans[phase.planIndex].count !== 1 ? 's' : ''} placed on it.
                Deleting will remove {phase.plans[phase.planIndex].count === 1 ? 'this marker' : 'these markers'}.
              </Alert>
            </>
          )}
        </DialogContent>
        <DialogActions>
          <Button onClick={() => handleMarkerWarningResponse('cancel')}>Cancel All</Button>
          <Button onClick={() => handleMarkerWarningResponse('skip')}>Skip</Button>
          <Button
            onClick={() => handleMarkerWarningResponse('delete')}
            color="error"
            variant="contained"
          >
            Delete
          </Button>
        </DialogActions>
      </Dialog>

      {/* Deleting progress dialog */}
      <Dialog open={phase.step === 'deleting'} maxWidth="sm" fullWidth>
        <DialogTitle>Deleting Floor Plans</DialogTitle>
        <DialogContent>
          {error && (
            <Alert severity="error" sx={{ mb: 2 }}>
              {error}
            </Alert>
          )}
          <Box sx={{ display: 'flex', alignItems: 'center', gap: 2 }}>
            <CircularProgress size={24} />
            <Typography>
              Deleting {phase.step === 'deleting' ? phase.progress : 0} of{' '}
              {phase.step === 'deleting' ? phase.total : 0}...
            </Typography>
          </Box>
          {phase.step === 'deleting' && (
            <LinearProgress
              variant="determinate"
              value={(phase.progress / phase.total) * 100}
              sx={{ mt: 2 }}
            />
          )}
        </DialogContent>
      </Dialog>

      {/* Complete dialog */}
      <Dialog open={phase.step === 'complete'} onClose={handleClose} maxWidth="sm" fullWidth>
        <DialogTitle>Delete Complete</DialogTitle>
        <DialogContent>
          {error && (
            <Alert severity="error" sx={{ mb: 2 }}>
              {error}
            </Alert>
          )}
          <Typography>
            {phase.step === 'complete' && phase.deleted > 0 && (
              <>Deleted {phase.deleted} floor plan{phase.deleted !== 1 ? 's' : ''}. </>
            )}
            {phase.step === 'complete' && phase.skipped > 0 && (
              <>Skipped {phase.skipped}.</>
            )}
          </Typography>
        </DialogContent>
        <DialogActions>
          <Button onClick={handleClose} variant="contained">
            Close
          </Button>
        </DialogActions>
      </Dialog>
    </>
  );
}
```
